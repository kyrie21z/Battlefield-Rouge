extends SceneTree
func _initialize() -> void:
	DirAccess.make_dir_recursive_absolute("res://.runtime/audio-before")
	SorrowSoundBank.get_sound(&"rain",true).save_to_wav("res://.runtime/audio-before/rain.wav")
	var hashes := {}
	for kind in SorrowSoundBank.KINDS:
		var hash := HashingContext.new()
		hash.start(HashingContext.HASH_SHA256)
		hash.update(SorrowSoundBank.get_sound(kind).data)
		hashes[kind] = hash.finish().hex_encode()
	var file := FileAccess.open("res://.runtime/audio-before/cue-hashes.json",FileAccess.WRITE)
	file.store_string(JSON.stringify(hashes,"  "))
	quit()
